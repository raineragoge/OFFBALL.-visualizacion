def generar_perfiles(df, X):
    probs = gmm_3_of.predict_proba(X)
    etiquetas = gmm_3_of.predict(X)

    salida = df[
        ["player_id", "season", "competition_id"]
    ].copy().reset_index(drop=True)

    salida["cluster_gmm_of"] = etiquetas
    salida["perfil_gmm_of"] = [
        nombres[int(c)] for c in etiquetas
    ]

    salida["prob_goleador"] = probs[:, cluster_goleador]
    salida["prob_asistente"] = probs[:, cluster_asistente]
    salida["prob_moderado"] = probs[:, cluster_moderado]
    salida["confianza_gmm_of"] = probs.max(axis=1)

        salida[
            ["prob_goleador",
             "prob_asistente",
             "prob_moderado"]
        ].sum(axis=1).sub(1).abs() < 1e-6
